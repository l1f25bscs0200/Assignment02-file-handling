#include <iostream>
#include <fstream>
using namespace std;

int main() {

    ifstream inFile("heart_rate.txt");

    if (!inFile) {
        cout << "heart_rate.txt not found!" << endl;
        return 1;
    }

    int rate[100];
    int count = 0, bad = 0;
    int x, reason = 0;
    char junk[50];

    while (true) {

        if (count == 100) {
            reason = 3;
            break;
        }

        if (inFile >> x) {

            if (x == -1) {
                reason = 1;
                break;
            }

            rate[count] = x;
            count++;

        }
        else if (inFile.eof()) {
            reason = 2;
            break;
        }
        else {
            inFile.clear();
            inFile >> junk;
            bad++;
        }
    }

    inFile.close();

    ofstream outFile("clean_heart_rate.txt");

    int total = 0;
    int highest = 0;
    int alerts = 0;
    int above = 0;

    for (int i = 0; i < count; i++) {

        outFile << rate[i] << endl;

        total += rate[i];

        if (i == 0 || rate[i] > highest) {
            highest = rate[i];
        }

        if (rate[i] > 100) {
            alerts++;
        }
    }

    outFile.close();

    cout << "Valid readings: " << count << endl;
    cout << "Junk skipped: " << bad << endl;

    if (count == 0) {
        cout << "No valid readings!" << endl;
    }
    else {

        double average = (double)total / count;

        for (int i = 0; i < count; i++) {
            if (rate[i] > average) {
                above++;
            }
        }

        cout << "Average: " << average << endl;
        cout << "Highest: " << highest << endl;
        cout << "Above average: " << above << endl;
        cout << "Alerts (above 100): "
             << alerts << endl;
    }

    cout << "Stopped because: ";

    if (reason == 1) {
        cout << "sentinel found";
    }
    else if (reason == 2) {
        cout << "end of file";
    }
    else {
        cout << "array full";
    }

    cout << endl;

    return 0;
}
