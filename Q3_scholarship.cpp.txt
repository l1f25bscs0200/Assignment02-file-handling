#include <iostream>
#include <fstream>
#include <cstring>
using namespace std;

int main() {
    char path[100];

    cout << "Enter the file name: ";
    cin.getline(path, 100);

    ifstream inFile(path);

    if (!inFile.is_open()) {
        cout << "Could not open " << path << "!" << endl;
        return 1;
    }

    int n;
    if (!(inFile >> n))
        n = 0;

    if (n > 30) {
        cout << "Too many students!" << endl;
        inFile.close();
        return 0;
    }

    inFile.ignore(100, '\n');

    ofstream outFile("scholarship.txt");

    int count = 0, eligible = 0;
    double total = 0, highest = -1, cgpa;
    char name[30], top[30] = "";

    while (count < n) {
        if (!inFile.getline(name, 30, ','))
            break;

        if (!(inFile >> cgpa))
            break;

        inFile.ignore(100, '\n');

        cout << "[" << name << "] CGPA: "
             << cgpa << endl;

        count++;
        total += cgpa;

        if (cgpa >= 3.50) {
            eligible++;
            outFile << name << ","
                    << cgpa << endl;
        }

        if (cgpa > highest) {
            highest = cgpa;
            strcpy(top, name);
        }
    }

    inFile.close();
    outFile.close();

    if (count < n) {
        cout << "Warning: only " << count
             << " of " << n
             << " records found." << endl;
    }

    if (count == 0) {
        cout << "No students found!" << endl;
        return 0;
    }

    cout << "Students: " << count << endl;
    cout << "Average CGPA: "
         << total / count << endl;

    cout << "Eligible for scholarship: "
         << eligible << endl;

    cout << "Top student: " << top
         << " (" << highest << ")" << endl;

    return 0;
}
